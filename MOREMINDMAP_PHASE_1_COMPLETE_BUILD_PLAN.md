```javascript
import puppeteer from 'puppeteer'
import { writeFile } from 'fs/promises'
import { mkdir } from 'fs/promises'
import path from 'path'
import { fileURLToPath } from 'url'

const __dirname = path.dirname(fileURLToPath(import.meta.url))

async function runSmokeTest() {
  console.log('[SMOKE TEST] Starting Puppeteer test...')
  
  try {
    // 1. Create temp folder
    const tempDir = path.join(__dirname, '.smoke-test')
    await mkdir(tempDir, { recursive: true })
    
    // 2. Launch Chromium
    console.log('[SMOKE TEST] Launching Chromium...')
    const startTime = Date.now()
    const browser = await puppeteer.launch({
      headless: true,
      args: ['--no-sandbox', '--disable-setuid-sandbox']
    })
    const launchTime = Date.now() - startTime
    console.log(`[SMOKE TEST] Chromium launched in ${launchTime}ms`)
    
    // 3. Create page
    console.log('[SMOKE TEST] Creating page...')
    const page = await browser.newPage()
    
    // 4. Set HTML content
    const htmlContent = `
      <!DOCTYPE html>
      <html>
      <head>
        <title>Puppeteer Smoke Test</title>
        <style>
          body { font-family: Arial, sans-serif; margin: 40px; }
          h1 { color: #333; }
          p { line-height: 1.6; }
        </style>
      </head>
      <body>
        <h1>Puppeteer Smoke Test</h1>
        <p>This is a test PDF generated by Puppeteer.</p>
        <p>Timestamp: ${new Date().toISOString()}</p>
        <p>If you see this, PDF generation is working correctly.</p>
      </body>
      </html>
    `
    
    await page.setContent(htmlContent)
    
    // 5. Generate PDF
    console.log('[SMOKE TEST] Generating PDF...')
    const pdfStartTime = Date.now()
    const pdfBuffer = await page.pdf({
      format: 'A4',
      margin: { top: 20, right: 20, bottom: 20, left: 20 }
    })
    const pdfTime = Date.now() - pdfStartTime
    console.log(`[SMOKE TEST] PDF generated in ${pdfTime}ms`)
    
    // 6. Save PDF
    const pdfPath = path.join(tempDir, 'smoke-test.pdf')
    await writeFile(pdfPath, pdfBuffer)
    const fileSizeKB = (pdfBuffer.length / 1024).toFixed(2)
    console.log(`[SMOKE TEST] PDF saved to ${pdfPath}`)
    console.log(`[SMOKE TEST] File size: ${fileSizeKB} KB`)
    
    // 7. Close browser
    console.log('[SMOKE TEST] Closing browser...')
    await browser.close()
    
    // 8. Report results
    console.log('\n✅ SMOKE TEST PASSED')
    console.log('─'.repeat(50))
    console.log(`Chromium Launch Time: ${launchTime}ms`)
    console.log(`PDF Generation Time: ${pdfTime}ms`)
    console.log(`PDF File Size: ${fileSizeKB} KB`)
    console.log(`Test PDF: ${pdfPath}`)
    console.log('─'.repeat(50))
    console.log('\n✅ Puppeteer is ready for production use')
    
  } catch (error) {
    console.error('\n❌ SMOKE TEST FAILED')
    console.error('Error:', error.message)
    process.exit(1)
  }
}

runSmokeTest()
```

## FORMSPREE ARCHIVAL INTEGRATION (NEW REQUIREMENT)

### Purpose

After Resend emails PDF to client + Darren, archive submission metadata to Formspree for backup record-keeping.

**Important:**
- Formspree is archival ONLY, not primary delivery
- Email delivery via Resend is professional channel
- Formspree failures must NOT block user experience
- All failures gracefully logged, no error bubbling

### Formspree Endpoint

```
https://formspree.io/f/mbdwjgvk
```

### Formspree Fields Required

```javascript
{
  name: string,                          // User's name
  email: string,                         // User's email
  assessmentType: "Mini Profile",        // Fixed value
  primaryPattern: string[],              // Top 2 dimension keys
  secondaryPattern: string[],            // Next 2 dimension keys
  promoCodeUsed: string || null,         // Promo code if used
  paymentStatus: "paid" || "free",       // Payment status
  completionTimestamp: ISO string,       // When assessment completed
  confidenceLevel: "high" || "medium" || "low",
  validityFlags: string[],               // Any validity flags
  reportGenerated: boolean,              // PDF generated successfully?
  emailDeliveryStatus: "success" || "failure" || "partial"  // Email status
}
```

### Implementation Location

New file: `/Users/rrg/moremindmap/utils/archiveService.js`

```javascript
/**
 * Archive Mini Profile submission metadata to Formspree
 * Non-blocking: failures logged but do not affect user experience
 */
export async function archiveSubmissionToFormspree(submissionData) {
  try {
    const payload = {
      name: submissionData.name,
      email: submissionData.email,
      assessmentType: "Mini Profile",
      primaryPattern: submissionData.primaryPattern.map(p => p.key).join(", "),
      secondaryPattern: submissionData.secondaryPattern.map(p => p.key).join(", "),
      promoCodeUsed: submissionData.promoCodeUsed || "none",
      paymentStatus: submissionData.paymentStatus || "unknown",
      completionTimestamp: new Date().toISOString(),
      confidenceLevel: submissionData.confidenceLevel,
      validityFlags: (submissionData.validityFlags || []).join(", "),
      reportGenerated: submissionData.reportGenerated,
      emailDeliveryStatus: submissionData.emailDeliveryStatus
    }
    
    const response = await fetch("https://formspree.io/f/mbdwjgvk", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    })
    
    if (!response.ok) {
      console.warn(
        `[ARCHIVE] Formspree archival failed (${response.status}). ` +
        `This is non-critical. User experience not affected.`
      )
      return { success: false, reason: `HTTP ${response.status}` }
    }
    
    console.log(`[ARCHIVE] Submission archived for ${submissionData.email}`)
    return { success: true }
    
  } catch (error) {
    // Silently log, do not throw
    console.warn(`[ARCHIVE] Formspree archival error: ${error.message}`)
    return { success: false, reason: error.message }
  }
}
```

```
  ↓

  ↓

  ├─→ HTML template
  ├─→ Chromium render
  ├─→ PDF output
  └─→ If fails: user sees HTML instead
  ↓
emailProfilePdf() (Resend)
  ├─→ Email to client
  ├─→ Email to Darren
  ├─→ Capture delivery status
  └─→ If fails: logged, user sees report anyway
  ↓

  ├─→ POST metadata to archive
  ├─→ No blocking on failure
  └─→ Logged only
  ↓
Response to frontend
  ├─→ User sees HTML report immediately
  ├─→ PDF emailed in background
  └─→ Archive recorded asynchronously
```
